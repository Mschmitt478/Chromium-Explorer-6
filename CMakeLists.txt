cmake_minimum_required(VERSION 3.22)
project(InternetExplorerXP VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets WebEngineWidgets PrintSupport Sql Network)
qt_standard_project_setup()
add_executable(iexplore src/main.cpp src/window.cpp src/dialogs.cpp src/engine.cpp src/store.cpp src/theme.cpp)
target_link_libraries(iexplore PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::PrintSupport Qt6::Sql Qt6::Network)
target_compile_definitions(iexplore PRIVATE EXPLORER_VERSION="${PROJECT_VERSION}")
qt_add_resources(iexplore artwork PREFIX "/" BASE assets FILES assets/explorer.svg)
install(TARGETS iexplore RUNTIME DESTINATION bin)
set(APP_ID io.github.Mschmitt478.ChromiumExplorer)
install(FILES packaging/${APP_ID}.desktop DESTINATION share/applications)
install(FILES packaging/${APP_ID}.metainfo.xml DESTINATION share/metainfo)
install(FILES assets/explorer.svg DESTINATION share/icons/hicolor/scalable/apps RENAME ${APP_ID}.svg)
foreach(size IN ITEMS 48 64 128 256)
    install(FILES assets/explorer-${size}.png
        DESTINATION share/icons/hicolor/${size}x${size}/apps RENAME ${APP_ID}.png)
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(store-test tests/store.cpp src/store.cpp)
    target_include_directories(store-test PRIVATE src)
    target_link_libraries(store-test PRIVATE Qt6::Sql)
    add_test(NAME storage-and-addresses COMMAND store-test)
    option(RUN_GUI_TESTS "Run tests requiring a desktop and WebEngine sandbox" OFF)
    if(RUN_GUI_TESTS)
        add_executable(engine-test tests/engine.cpp src/engine.cpp src/window.cpp src/dialogs.cpp src/store.cpp src/theme.cpp)
        target_include_directories(engine-test PRIVATE src)
        target_link_libraries(engine-test PRIVATE Qt6::WebEngineWidgets Qt6::Network Qt6::PrintSupport Qt6::Sql)
        target_compile_definitions(engine-test PRIVATE EXPLORER_VERSION="${PROJECT_VERSION}")
        add_test(NAME chromium-navigation COMMAND engine-test)
        set_tests_properties(chromium-navigation PROPERTIES TIMEOUT 65)
    endif()
endif()
